   102	    # Serves the given [status, JSON] responses in order from a loopback HTTP
   103	    # server, repeating the last, and returns each request's DOI and
   104	    # Authorization header.
   105	    def with_provider(*responses)
   106	      requests = []
   107	      app = lambda do |env|
   108	        requests << [Rack::Request.new(env).params["doi"], env["HTTP_AUTHORIZATION"]]
   109	        status, body = responses.size > 1 ? responses.shift : responses.first
   110	        [status, { "Content-Type" => "application/json" }, [body.to_json]]
   111	      end
   112	
   113	      server = Puma::Server.new(app, Puma::Events.strings)
   114	      server.add_tcp_listener("127.0.0.1", 0)
   115	      server.run
   116	
   117	      settings = Rails.application.secrets.citation_provider
   118	      Rails.application.secrets.citation_provider = settings.merge(url: "http://127.0.0.1:#{server.connected_ports.first}")
   119	      yield
   120	      requests
   121	    ensure
   122	      Rails.application.secrets.citation_provider = settings if settings
   123	      server&.stop(true)
   124	    end
