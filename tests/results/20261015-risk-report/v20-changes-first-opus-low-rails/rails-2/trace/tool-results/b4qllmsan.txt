# A work identified by its DOI, shared by every reader who cites it. Readers'
# own annotations stay on their notes. Its metadata, and the state of the latest
# request for it, are shared too.
class Citation < ApplicationRecord
  has_many :notes, foreign_key: :citation_doi

  enum :enrichment_status, { pending: 0, completed: 1, failed: 2 }, prefix: :enrichment

  validates :doi, format: { with: %r{\A10\.\d{4,9}/\S+\z} }

  # Marks the citation pending and queues a metadata lookup, unless one is
  # already pending. Returns whether a lookup was queued. The job is queued under
  # the row lock, so a failed enqueue rolls back the pending state.
  def request_enrichment
    with_lock do
      if enrichment_pending?
        false
      else
        update!(enrichment_status: :pending, enrichment_error: nil)
        CitationEnrichmentJob.perform_later(self)
        true
      end
    end
  end

  # Asks the provider for this DOI's metadata. A permanent refusal marks the
  # enrichment failed; an unavailable provider raises so the job can retry.
  def enrich
    metadata = Provider.new.lookup(doi)
    record_enrichment(title: metadata.title, source_url: metadata.source_url, enrichment_status: :completed, enrichment_error: nil)
  rescue Provider::Error => error
    fail_enrichment(error.message)
  end

  def fail_enrichment(reason)
    record_enrichment(enrichment_status: :failed, enrichment_error: reason)
  end

  private
    # The job can load this citation before the request that queued it commits.
    # Taking the row lock waits for that commit and reloads, so the result is
    # written even when it equals the values loaded earlier.
    def record_enrichment(attributes)
      with_lock { update!(attributes) }
    end
end
# A Markdown file of one notebook's notes, written by ExportJob. The reader can
# cancel it while it is queued or working; the worker checks between notes and
# stops. A completed file can be downloaded until it expires.
class Export < ApplicationRecord
  AVAILABLE_FOR = 24.hours

  belongs_to :notebook
  has_one_attached :file

  enum :status, { queued: 0, working: 1, completed: 2, cancelled: 3 }

  scope :newest_first, -> { order(created_at: :desc, id: :desc) }

  def cancellable?
    queued? || working?
  end

  # Cancels a queued or working export. Returns whether it was cancelled.
  def cancel
    with_lock do
      if cancellable?
        update!(status: :cancelled)
      else
        false
      end
    end
  end

  # Writes the notebook's notes, oldest first, under its export heading. Progress
  # is saved after each note. A cancelled export stops before the next note and
  # keeps no file.
  def generate
    notes = notebook.notes.with_rich_text_body.order(:created_at, :id).to_a
    return unless start(notes.size)

    fragments = ["# #{notebook.export_heading || notebook.name}\n"]
    notes.each_with_index do |note, index|
      fragments << markdown_for(note)
      return unless record_progress(index + 1)
    end

    finish(fragments.sum)
  end

  # The authoritative check for downloading the file.
  def downloadable?
    completed? && expires_at.future?
  end

  private
    # A retried job starts again from the first note.
    def start(total)
      with_lock do
        update!(status: :working, processed_notes_count: 0, total_notes_count: total) if cancellable?
      end
    end

    def record_progress(count)
      with_lock do
        update!(processed_notes_count: count) unless cancelled?
      end
    end

    # The file is stored before the export is marked completed, so a completed
    # export always has one.
    def finish(markdown)
      blob = ActiveStorage::Blob.create_and_upload!(io: StringIO.new(markdown),
        filename: "#{notebook.slug}.md", content_type: "text/markdown")

      completed = with_lock do
        update!(file: blob, status: :completed, expires_at: AVAILABLE_FOR.from_now) unless cancelled?
      end

      blob.purge unless completed
    end

    def markdown_for(note)
      body = note.body.to_plain_text.strip
      markdown = "\n## #{note.title.squish}\n"
      markdown += "\n#{body}\n" if body.present?
      markdown
    end
end
require "csv"

# A CSV of notes for one notebook. A preview reads and validates the rows without
# saving anything. Applying validates them again and saves the import with every
# row's note, or saves nothing.
class Import < ApplicationRecord
  HEADERS = %w[ title body_html doi annotation reminder_epoch ]
  MAXIMUM_SIZE = 100.kilobytes
  MAXIMUM_ROWS = 100

  Row = Struct.new(:line, :note, :annotation)

  belongs_to :notebook
  has_many :notes, dependent: :nullify

  validate :csv_is_importable, on: :create

  # Reads one byte past the limit, which is enough to reject a larger file.
  def file=(upload)
    self.csv = String.new(upload.read(MAXIMUM_SIZE + 1).to_s, encoding: Encoding::UTF_8)
  end

  def csv
    @csv.to_s
  end

  def csv=(text)
    @csv = text.to_s
    @rows = nil
  end

  def rows
    parse unless @rows
    @rows
  end

  # The reader's annotations by source line.
  def annotations
    rows.each_with_object(ActiveSupport::HashWithIndifferentAccess.new) do |row, annotations|
      annotations[row.line] = row.annotation
    end
  end

  # Validates the file and every row without saving. Returns whether the import
  # can be applied, leaving the errors on the import and each row's note.
  def preview
    valid? && rows.map { |row| row.note.valid? }.all?
  end

  def apply
    return false if invalid?

    transaction do
      save!
      rows.each do |row|
        return false unless row.note.save
      end
    end

    true
  end

  def receipt
    CSV.generate do |csv|
      csv << %w[ title doi ]
      notes.includes(:citation).order(:id).each do |note|
        citation = note.citation
        csv << [ note.title, citation && citation[:id] ]
      end
    end
  end

  private
    def parse
      @rows, @headers, @problem = [], nil, nil
      return if csv.bytesize > MAXIMUM_SIZE || !csv.valid_encoding?

      reader = CSV.new(csv.delete_prefix("\uFEFF").encode(universal_newline: true))
      citations = {}
      next_line = 1

      reader.each do |fields|
        line, next_line = next_line, next_line + reader.line.count("\n")

        if @headers.nil?
          @headers = fields.map { |field| field.to_s.strip }
          break if @headers != HEADERS
        elsif fields.size != HEADERS.size && fields.any?
          @problem = "line #{line} has #{fields.size} fields instead of #{HEADERS.size}"
          break
        elsif fields.any?
          @rows << build_row(line, fields, citations)
          break if @rows.size > MAXIMUM_ROWS
        end
      end
    rescue CSV::MalformedCSVError => error
      @problem = "it is not valid CSV (#{error.message})"
    end

    # Rows naming the same DOI share one citation, so it is created only once.
    def build_row(line, fields, citations)
      title, body_html, doi, annotation, reminder_epoch = fields.map { |field| field.to_s.strip }
      citation = citations[doi] ||= Citation.find_or_initialize_by(doi: doi) if doi.present?

      note = Note.new(notebook: notebook, import: self, title: title, body: body_html, citation: citation,
        annotation: annotation.presence, reminder_epoch: reminder_epoch.presence)

      Row.new(line, note, annotation.presence)
    end

    def csv_is_importable
      rows

      if csv.empty?
        errors.add(:csv, :blank)
      elsif csv.bytesize > MAXIMUM_SIZE
        errors.add(:csv, "must be #{ActiveSupport::NumberHelper.number_to_human_size(MAXIMUM_SIZE)} or smaller")
      elsif !csv.valid_encoding?
        errors.add(:csv, "must be UTF-8 text")
      elsif @problem
        errors.add(:csv, "cannot be read because #{@problem}")
      elsif @headers != HEADERS
        errors.add(:csv, "must start with the header row #{HEADERS.join(",")}")
      elsif rows.empty?
        errors.add(:csv, "must contain at least one note")
      elsif rows.size > MAXIMUM_ROWS
        errors.add(:csv, "must contain #{MAXIMUM_ROWS} notes or fewer")
      end
    end
end
class Note < ApplicationRecord
  HIGHLIGHT_COLORS = %w[ yellow green blue pink ]
  MAXIMUM_ATTACHMENT_SIZE = 1.megabyte

  belongs_to :notebook
  belongs_to :import, optional: true
  belongs_to :citation, optional: true, foreign_key: :citation_doi, validate: true
  has_rich_text :body

  enum status: { draft: 0, ready: 1 }, _prefix: true

  serialize :tags, JSON
  store_accessor :settings, :color
  alias_attribute :highlight_color, :color

  validates :title, presence: true
  validates :color, inclusion: { in: HIGHLIGHT_COLORS }, allow_blank: true
  validates :reminder_epoch, numericality: { only_integer: true, greater_than: 0, less_than: 10_000_000_000 }, allow_nil: true
  validate :attachments_are_small

  scope :newest_first, -> { order(created_at: :desc, id: :desc) }
  scope :titled_like, ->(query) { where("notes.title ILIKE ?", "%#{sanitize_sql_like(query)}%") }

  def doi
    citation_doi
  end

  # Cites the shared citation with this exact DOI, ignoring surrounding whitespace.
  # A blank DOI removes the citation from the note.
  def doi=(doi)
    doi = doi.to_s.strip
    self.citation = (Citation.find_or_initialize_by(doi: doi) if doi.present?)
  end

  # The reminder time in the reader's time zone.
  def reminder_at
    Time.find_zone!(notebook.user.time_zone).at(reminder_epoch) if reminder_epoch
  end

  # Twenty-four elapsed hours after the reminder time, for postponing it a day.
  # The result keeps the reminder's UTC offset, so across a daylight saving
  # change its wall-clock time differs from the reader's local time.
  def next_day_reminder_at
    reminder_at.to_time + 86_400
  end

  def reminder_scheduled?
    reminder_job_id.present? && reminder_sent_at.nil?
  end

  # Schedules an email reminder at the given time in seconds since 1970, by
  # default the stored one, such as an imported reminder. It replaces any
  # earlier reminder for this note: the earlier job sends nothing when it runs.
  # Returns whether the reminder was scheduled. The job is queued under the row
  # lock, so a failed enqueue leaves the previous reminder in place.
  def schedule_reminder(epoch = reminder_epoch)
    with_lock do
      self.reminder_epoch = epoch

      if reminder_epoch && valid?
        job = NoteReminderJob.set(wait_until: reminder_epoch).perform_later(self)
        update!(reminder_job_id: job.job_id, reminder_sent_at: nil)
      else
        errors.add(:base, "Choose a date and time for the reminder") if reminder_epoch.nil?
        restore_attributes([:reminder_epoch])
        false
      end
    end
  end

  def postpone_reminder
    reminder_scheduled? && schedule_reminder(next_day_reminder_at.to_i)
  end

  # Sends the reminder if the given job is still the one scheduled for it.
  # Taking the row lock waits for the request that queued the job to commit.
  def send_reminder(job_id)
    with_lock do
      if reminder_job_id == job_id && reminder_scheduled?
        update!(reminder_sent_at: Time.current)
        ReminderMailer.with(note: self).note_reminder.deliver_now
      end
    end
  end

  def tag_list
    tags.join(", ")
  end

  def tag_list=(list)
    self.tags = list.to_s.split(",").map(&:strip).reject(&:blank?).uniq
  end

  private
    def attachments_are_small
      if body.body&.attachables.to_a.grep(ActiveStorage::Blob).any? { |blob| blob.byte_size > MAXIMUM_ATTACHMENT_SIZE }
        errors.add(:body, "attachments must be #{ActiveSupport::NumberHelper.number_to_human_size(MAXIMUM_ATTACHMENT_SIZE)} or smaller")
      end
    end
end
class Notebook < ApplicationRecord
  belongs_to :user
  has_many :notes, dependent: :destroy
  has_many :imports, dependent: :destroy
  has_many :exports, dependent: :destroy

  serialize :preferences
  attr_readonly :slug

  before_validation :assign_slug, on: :create

  validates :name, presence: true

  def export_heading
    preferences.to_h["export_heading"]
  end

  def export_heading=(heading)
    self.preferences = preferences.to_h.merge("export_heading" => heading.to_s.strip.presence).compact
  end

  # Queues a Markdown export. The export is saved before its job is queued, so
  # the worker can always load it, and removed again if Redis refuses the job.
  def request_export
    export = exports.create!
    export.update!(job_id: ExportJob.perform_async(export.id, { format: "markdown" }))
    export
  rescue Redis::BaseError
    export&.destroy
    raise
  end

  # The number of notes, in any status, created on each UTC date, newest date
  # first, as [date, count] pairs. Dates without notes are omitted. Only notes
  # that still exist are counted, so deleting a note lowers its date's count.
  # Timestamps are stored in UTC, so their date part is the UTC date. The raw
  # select returns that date as PostgreSQL's ISO 8601 text.
  def daily_note_counts
    days = notes.select("DATE(notes.created_at) AS day, COUNT(*) AS notes_count").group("day").order("day DESC")
    Note.connection.select_rows(days).map { |day, count| [Date.iso8601(day), count] }
  end

  private
    def assign_slug
      self.slug = name.to_s.parameterize.presence || "notebook"
    end
end
class User < ApplicationRecord
  # Keep digest reservations past the end of their UTC day.
  DIGEST_RESERVATIONS_KEPT_FOR = 2.days

  has_secure_password

  has_many :notebooks, dependent: :destroy
  has_many :notes, through: :notebooks

  before_validation { self.email = email.to_s.strip.downcase }

  validates :email, presence: true, uniqueness: true
  validates :time_zone, inclusion: { in: ActiveSupport::TimeZone.all.map(&:name) }

  def self.authenticate(email, password)
    find_by(email: email.to_s.strip.downcase)&.authenticate(password)
  end

  # Reads a date and time the reader entered, such as "2026-10-31T09:00", in
  # their time zone. Returns nil when it is blank or not a valid time.
  def local_time(text)
    Time.find_zone!(time_zone).iso8601(text.to_s) if text.present?
  rescue ArgumentError
    nil
  end

  # Queues an email digest of ready notes updated today (UTC), unless the reader
  # has already requested one for that UTC day. Returns whether a digest was queued.
  #
  # A Redis set of reader IDs for each day holds the reservations. If the job
  # cannot be queued, the reservation is released and the error raised, so the
  # reader can ask again. This prevents repeated requests, not repeated
  # deliveries: a job that Sidekiq retries can send its digest more than once.
  def request_digest
    requested_at = Time.current.in_time_zone(time_zone)
    key = "digest_requests:#{requested_at.utc.to_date.iso8601}"

    if Sidekiq.redis { |redis| redis.sadd(key, id) }
      begin
        Sidekiq.redis { |redis| redis.expire(key, DIGEST_RESERVATIONS_KEPT_FOR.to_i) }
        DigestJob.perform_later(self, requested_at, reading_completion)
      rescue StandardError
        Sidekiq.redis { |redis| redis.srem(key, id) }
        raise
      end

      true
    else
      false
    end
  end

  # The share of the reader's notes that are ready, from 0 to 1.
  def reading_completion
    total = notes.count
    total.zero? ? BigDecimal("0") : (BigDecimal(notes.status_ready.count) / total).round(4)
  end
end
# The configured citation metadata service. It answers GET /works?doi=... with a
# JSON title and source URL, using HTTP basic authentication. Settings come from
# config/secrets.yml, which reads them from the environment.
class Citation::Provider
  # The provider answered, but cannot supply metadata for this DOI.
  class Error < StandardError; end

  # The provider cannot answer now. Asking again later may succeed.
  class Unavailable < StandardError; end

  Metadata = Struct.new(:title, :source_url)

  def initialize(settings = Rails.application.secrets.citation_provider.to_h)
    @url, @user, @password = settings.values_at(:url, :user, :password)
  end

  def lookup(doi)
    raise Error, "No citation provider is configured" if @url.blank?

    response = connection.get("works", doi: doi)

    case response.status
    when 200 then metadata_from(response.body)
    when 401, 403 then raise Error, "The provider rejected the configured credentials (HTTP #{response.status})"
    when 404 then raise Error, "The provider has no record of this DOI"
    when 429, 500..599 then raise Unavailable, "The provider is unavailable (HTTP #{response.status})"
    else raise Error, "The provider gave an unexpected response (HTTP #{response.status})"
    end
  rescue Faraday::ConnectionFailed, Faraday::TimeoutError => error
    raise Unavailable, "The provider could not be reached (#{error.message})"
  end

  private
    # Retries a GET once on 503 before reporting the provider unavailable.
    def connection
      Faraday.new(url: @url, request: { open_timeout: 2, timeout: 5 }) do |faraday|
        faraday.request :retry, max: 1, retry_statuses: [503]
        faraday.request :basic_auth, @user, @password
      end
    end

    def metadata_from(body)
      attributes = JSON.parse(body)
      title, source_url = attributes["title"], attributes["url"] if attributes.is_a?(Hash)

      if title.is_a?(String) && title.present? && web_url?(source_url)
        Metadata.new(title.strip, source_url)
      else
        raise Error, "The provider's response did not include a title and web address"
      end
    rescue JSON::ParserError
      raise Error, "The provider's response was not valid JSON"
    end

    def web_url?(url)
      uri = URI.parse(url.to_s)
      uri.is_a?(URI::HTTP) && uri.host.present?
    rescue URI::InvalidURIError
      false
    end
end
