   158	  private
   159	    # Serves the given [status, JSON] responses in order from a loopback HTTP
   160	    # server, repeating the last, and returns each request's DOI and
   161	    # Authorization header.
   162	    def with_provider(*responses)
   163	      requests = []
   164	      app = lambda do |env|
   165	        requests << [Rack::Request.new(env).params["doi"], env["HTTP_AUTHORIZATION"]]
   166	        status, body = responses.size > 1 ? responses.shift : responses.first
   167	        [status, { "Content-Type" => "application/json" }, [body.to_json]]
   168	      end
   169	
   170	      server = Puma::Server.new(app, Puma::Events.strings)
   171	      server.add_tcp_listener("127.0.0.1", 0)
   172	      server.run
   173	
   174	      settings = Rails.application.secrets.citation_provider
   175	      Rails.application.secrets.citation_provider = settings.merge(url: "http://127.0.0.1:#{server.connected_ports.first}")
   176	      yield
   177	      requests
   178	    ensure
   179	      Rails.application.secrets.citation_provider = settings if settings
   180	      server&.stop(true)
   181	    end
   182	
   183	    def basic_auth(user, password)
   184	      "Basic #{Base64.strict_encode64("#{user}:#{password}")}"
   185	    end

   775	    nine_eastern = Time.utc(2026, 10, 31, 13).to_i

   779	
   780	    follow_redirect!
   781	    assert_select "#reminder", /Scheduled\s+for\s+Sat 31 Oct 2026, 09:00 \(UTC-04:00\)/
   782	    # Daylight saving time ends before the next day; the preview keeps the offset.
   783	    assert_select "#next_day_reminder time[datetime=?]", "2026-11-01T09:00:00-04:00"
   784	
   785	    perform_enqueued_jobs
   786	    mail = ActionMailer::Base.deliveries.sole
   787	    assert_equal ["reader@example.com"], mail.to
   788	    assert_equal "Reminder: Montaigne on friendship", mail.subject

   790	    assert_includes mail.body.to_s, note_url(note)
