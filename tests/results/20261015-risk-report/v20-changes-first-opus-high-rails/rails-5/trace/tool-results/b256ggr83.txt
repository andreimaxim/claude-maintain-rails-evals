=== app/models/citation.rb
     1	# A work identified by its DOI, shared by every reader who cites it. Readers'
     2	# own annotations stay on their notes. Its metadata, and the state of the latest
     3	# request for it, are shared too.
     4	class Citation < ApplicationRecord
     5	  has_many :notes, foreign_key: :citation_doi
     6	
     7	  enum :enrichment_status, { pending: 0, completed: 1, failed: 2 }, prefix: :enrichment
     8	
     9	  validates :doi, format: { with: %r{\A10\.\d{4,9}/\S+\z} }
    10	
    11	  # Marks the citation pending and queues a metadata lookup, unless one is
    12	  # already pending. Returns whether a lookup was queued. The job is queued under
    13	  # the row lock, so a failed enqueue rolls back the pending state.
    14	  def request_enrichment
    15	    with_lock do
    16	      if enrichment_pending?
    17	        false
    18	      else
    19	        update!(enrichment_status: :pending, enrichment_error: nil)
    20	        CitationEnrichmentJob.perform_later(self)
    21	        true
    22	      end
    23	    end
    24	  end
    25	
    26	  # Asks the provider for this DOI's metadata. A permanent refusal marks the
    27	  # enrichment failed; an unavailable provider raises so the job can retry.
    28	  def enrich
    29	    metadata = Provider.new.lookup(doi)
    30	    record_enrichment(title: metadata.title, source_url: metadata.source_url, enrichment_status: :completed, enrichment_error: nil)
    31	  rescue Provider::Error => error
    32	    fail_enrichment(error.message)
    33	  end
    34	
    35	  def fail_enrichment(reason)
    36	    record_enrichment(enrichment_status: :failed, enrichment_error: reason)
    37	  end
    38	
    39	  private
    40	    # The job can load this citation before the request that queued it commits.
    41	    # Taking the row lock waits for that commit and reloads, so the result is
    42	    # written even when it equals the values loaded earlier.
    43	    def record_enrichment(attributes)
    44	      with_lock { update!(attributes) }
    45	    end
    46	end
=== app/models/citation/provider.rb
     1	# The configured citation metadata service. It answers GET /works?doi=... with a
     2	# JSON title and source URL, using HTTP basic authentication. Settings come from
     3	# config/secrets.yml, which reads them from the environment.
     4	class Citation::Provider
     5	  # The provider answered, but cannot supply metadata for this DOI.
     6	  class Error < StandardError; end
     7	
     8	  # The provider cannot answer now. Asking again later may succeed.
     9	  class Unavailable < StandardError; end
    10	
    11	  Metadata = Struct.new(:title, :source_url)
    12	
    13	  def initialize(settings = Rails.application.secrets.citation_provider.to_h)
    14	    @url, @user, @password = settings.values_at(:url, :user, :password)
    15	  end
    16	
    17	  def lookup(doi)
    18	    raise Error, "No citation provider is configured" if @url.blank?
    19	
    20	    response = connection.get("works", doi: doi)
    21	
    22	    case response.status
    23	    when 200 then metadata_from(response.body)
    24	    when 401, 403 then raise Error, "The provider rejected the configured credentials (HTTP #{response.status})"
    25	    when 404 then raise Error, "The provider has no record of this DOI"
    26	    when 429, 500..599 then raise Unavailable, "The provider is unavailable (HTTP #{response.status})"
    27	    else raise Error, "The provider gave an unexpected response (HTTP #{response.status})"
    28	    end
    29	  rescue Faraday::ConnectionFailed, Faraday::TimeoutError => error
    30	    raise Unavailable, "The provider could not be reached (#{error.message})"
    31	  end
    32	
    33	  private
    34	    # Retries a GET once on 503 before reporting the provider unavailable.
    35	    def connection
    36	      Faraday.new(url: @url, request: { open_timeout: 2, timeout: 5 }) do |faraday|
    37	        faraday.request :retry, max: 1, retry_statuses: [503]
    38	        faraday.request :basic_auth, @user, @password
    39	      end
    40	    end
    41	
    42	    def metadata_from(body)
    43	      attributes = JSON.parse(body)
    44	      title, source_url = attributes["title"], attributes["url"] if attributes.is_a?(Hash)
    45	
    46	      if title.is_a?(String) && title.present? && web_url?(source_url)
    47	        Metadata.new(title.strip, source_url)
    48	      else
    49	        raise Error, "The provider's response did not include a title and web address"
    50	      end
    51	    rescue JSON::ParserError
    52	      raise Error, "The provider's response was not valid JSON"
    53	    end
    54	
    55	    def web_url?(url)
    56	      uri = URI.parse(url.to_s)
    57	      uri.is_a?(URI::HTTP) && uri.host.present?
    58	    rescue URI::InvalidURIError
    59	      false
    60	    end
    61	end
=== app/models/export.rb
     1	# A Markdown file of one notebook's notes, written by ExportJob. The reader can
     2	# cancel it while it is queued or working; the worker checks between notes and
     3	# stops. A completed file can be downloaded until it expires.
     4	class Export < ApplicationRecord
     5	  AVAILABLE_FOR = 24.hours
     6	
     7	  belongs_to :notebook
     8	  has_one_attached :file
     9	
    10	  enum :status, { queued: 0, working: 1, completed: 2, cancelled: 3 }
    11	
    12	  scope :newest_first, -> { order(created_at: :desc, id: :desc) }
    13	
    14	  def cancellable?
    15	    queued? || working?
    16	  end
    17	
    18	  # Cancels a queued or working export. Returns whether it was cancelled.
    19	  def cancel
    20	    with_lock do
    21	      if cancellable?
    22	        update!(status: :cancelled)
    23	      else
    24	        false
    25	      end
    26	    end
    27	  end
    28	
    29	  # Writes the notebook's notes, oldest first, under its export heading. Progress
    30	  # is saved after each note. A cancelled export stops before the next note and
    31	  # keeps no file.
    32	  def generate
    33	    notes = notebook.notes.with_rich_text_body.order(:created_at, :id).to_a
    34	    return unless start(notes.size)
    35	
    36	    fragments = ["# #{notebook.export_heading || notebook.name}\n"]
    37	    notes.each_with_index do |note, index|
    38	      fragments << markdown_for(note)
    39	      return unless record_progress(index + 1)
    40	    end
    41	
    42	    finish(fragments.sum)
    43	  end
    44	
    45	  # The authoritative check for downloading the file.
    46	  def downloadable?
    47	    completed? && expires_at.future?
    48	  end
    49	
    50	  private
    51	    # A retried job starts again from the first note.
    52	    def start(total)
    53	      with_lock do
    54	        update!(status: :working, processed_notes_count: 0, total_notes_count: total) if cancellable?
    55	      end
    56	    end
    57	
    58	    def record_progress(count)
    59	      with_lock do

     3	# A CSV of notes for one notebook. A preview reads and validates the rows without
     4	# saving anything. Applying validates them again and saves the import with every
     5	# row's note, or saves nothing.
     6	class Import < ApplicationRecord
     7	  HEADERS = %w[ title body_html doi annotation reminder_epoch ]
     8	  MAXIMUM_SIZE = 100.kilobytes
     9	  MAXIMUM_ROWS = 100
    10	
    11	  Row = Struct.new(:line, :note, :annotation)
    12	
    13	  belongs_to :notebook
    14	  has_many :notes, dependent: :nullify
    15	
    16	  validate :csv_is_importable, on: :create
    17	
    18	  # Reads one byte past the limit, which is enough to reject a larger file.
    19	  def file=(upload)
    20	    self.csv = String.new(upload.read(MAXIMUM_SIZE + 1).to_s, encoding: Encoding::UTF_8)
    21	  end
    22	
    23	  def csv
    24	    @csv.to_s
    25	  end
    26	
    27	  def csv=(text)
    28	    @csv = text.to_s
    29	    @rows = nil
    30	  end
    31	
    32	  def rows
    33	    parse unless @rows
    34	    @rows
    35	  end
    36	
    37	  # The reader's annotations by source line.
    38	  def annotations
    39	    rows.each_with_object(ActiveSupport::HashWithIndifferentAccess.new) do |row, annotations|
    40	      annotations[row.line] = row.annotation
    41	    end
    42	  end
    43	
    44	  # Validates the file and every row without saving. Returns whether the import
    45	  # can be applied, leaving the errors on the import and each row's note.
    46	  def preview
    47	    valid? && rows.map { |row| row.note.valid? }.all?
    48	  end
    49	
    50	  def apply
    51	    return false if invalid?
    52	
    53	    transaction do
    54	      save!
    55	      rows.each do |row|
    56	        return false unless row.note.save
    57	      end
    58	    end
    59	
    60	    true
    61	  end
    62	
    63	  def receipt
    64	    CSV.generate do |csv|
    65	      csv << %w[ title doi ]
    66	      notes.includes(:citation).order(:id).each do |note|
    67	        citation = note.citation
    68	        csv << [ note.title, citation && citation[:id] ]
    69	      end
    70	    end
    71	  end
    72	
    73	  private
    74	    def parse
    75	      @rows, @headers, @problem = [], nil, nil
    76	      return if csv.bytesize > MAXIMUM_SIZE || !csv.valid_encoding?
    77	
    78	      reader = CSV.new(csv.delete_prefix("\uFEFF").encode(universal_newline: true))
    79	      citations = {}
    80	      next_line = 1
    81	
    82	      reader.each do |fields|
    83	        line, next_line = next_line, next_line + reader.line.count("\n")
    84	
    85	        if @headers.nil?
    86	          @headers = fields.map { |field| field.to_s.strip }
    87	          break if @headers != HEADERS
    88	        elsif fields.size != HEADERS.size && fields.any?
    89	          @problem = "line #{line} has #{fields.size} fields instead of #{HEADERS.size}"
    90	          break
    91	        elsif fields.any?
    92	          @rows << build_row(line, fields, citations)
    93	          break if @rows.size > MAXIMUM_ROWS
    94	        end
    95	      end
    96	    rescue CSV::MalformedCSVError => error
    97	      @problem = "it is not valid CSV (#{error.message})"
    98	    end
    99	
   100	    # Rows naming the same DOI share one citation, so it is created only once.
   101	    def build_row(line, fields, citations)
   102	      title, body_html, doi, annotation, reminder_epoch = fields.map { |field| field.to_s.strip }
   103	      citation = citations[doi] ||= Citation.find_or_initialize_by(doi: doi) if doi.present?
   104	
   105	      note = Note.new(notebook: notebook, import: self, title: title, body: body_html, citation: citation,
   106	        annotation: annotation.presence, reminder_epoch: reminder_epoch.presence)
   107	
   108	      Row.new(line, note, annotation.presence)
   109	    end
   110	
   111	    def csv_is_importable
   112	      rows
   113	
   114	      if csv.empty?
   115	        errors.add(:csv, :blank)
   116	      elsif csv.bytesize > MAXIMUM_SIZE
   117	        errors.add(:csv, "must be #{ActiveSupport::NumberHelper.number_to_human_size(MAXIMUM_SIZE)} or smaller")
   118	      elsif !csv.valid_encoding?
   119	        errors.add(:csv, "must be UTF-8 text")
   120	      elsif @problem
   121	        errors.add(:csv, "cannot be read because #{@problem}")
   122	      elsif @headers != HEADERS
   123	        errors.add(:csv, "must start with the header row #{HEADERS.join(",")}")
   124	      elsif rows.empty?
   125	        errors.add(:csv, "must contain at least one note")
   126	      elsif rows.size > MAXIMUM_ROWS
   127	        errors.add(:csv, "must contain #{MAXIMUM_ROWS} notes or fewer")
   128	      end
   129	    end
   130	end

     1	class User < ApplicationRecord
     2	  # Keep digest reservations past the end of their UTC day.
     3	  DIGEST_RESERVATIONS_KEPT_FOR = 2.days
     4

    10	  before_validation { self.email = email.to_s.strip.downcase }
    11	
    12	  validates :email, presence: true, uniqueness: true
    13	  validates :time_zone, inclusion: { in: ActiveSupport::TimeZone.all.map(&:name) }
    14	
    15	  def self.authenticate(email, password)
    16	    find_by(email: email.to_s.strip.downcase)&.authenticate(password)
    17	  end
    18	
    19	  # Reads a date and time the reader entered, such as "2026-10-31T09:00", in
    20	  # their time zone. Returns nil when it is blank or not a valid time.
    21	  def local_time(text)
    22	    Time.find_zone!(time_zone).iso8601(text.to_s) if text.present?
    23	  rescue ArgumentError

    25	  end
    26	
    27	  # Queues an email digest of ready notes updated today (UTC), unless the reader
    28	  # has already requested one for that UTC day. Returns whether a digest was queued.
    29	  #
    30	  # A Redis set of reader IDs for each day holds the reservations. If the job
    31	  # cannot be queued, the reservation is released and the error raised, so the
    32	  # reader can ask again. This prevents repeated requests, not repeated
    33	  # deliveries: a job that Sidekiq retries can send its digest more than once.
    34	  def request_digest
    35	    requested_at = Time.current.in_time_zone(time_zone)
    36	    key = "digest_requests:#{requested_at.utc.to_date.iso8601}"
    37	
    38	    if Sidekiq.redis { |redis| redis.sadd(key, id) }
    39	      begin
    40	        Sidekiq.redis { |redis| redis.expire(key, DIGEST_RESERVATIONS_KEPT_FOR.to_i) }
    41	        DigestJob.perform_later(self, requested_at, reading_completion)
    42	      rescue StandardError
    43	        Sidekiq.redis { |redis| redis.srem(key, id) }
    44	        raise
    45	      end
    46	
    47	      true
    48	    else
    49	      false
    50	    end
    51	  end
    52	
    53	  # The share of the reader's notes that are ready, from 0 to 1.
    54	  def reading_completion
    55	    total = notes.count
    56	    total.zero? ? BigDecimal("0") : (BigDecimal(notes.status_ready.count) / total).round(4)
    57	  end
